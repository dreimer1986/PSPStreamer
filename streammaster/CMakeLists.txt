cmake_minimum_required(VERSION 3.16)
set(STREAMMASTER_BOARD "onju-v3" CACHE STRING "onju-v3 or generic")
if(NOT STREAMMASTER_BOARD STREQUAL "onju-v3" AND NOT STREAMMASTER_BOARD STREQUAL "generic")
    message(FATAL_ERROR "Unknown STREAMMASTER_BOARD")
endif()
option(STREAMMASTER_O3 "Experimental O3 comparison build" OFF)
option(STREAMMASTER_BLUETOOTH "Experimental external USB Bluetooth build" OFF)
option(STREAMMASTER_QIO "QIO comparison label (requires sdkconfig.qio)" OFF)
option(STREAMMASTER_FLASH80 "80 MHz QIO comparison (requires sdkconfig.flash80)" OFF)
option(STREAMMASTER_HOT_IRAM "Isolated transport IRAM placement comparison" OFF)
set(STREAMMASTER_VERSION "0.3.19")
set(PROJECT_VER "${STREAMMASTER_VERSION}")
if(STREAMMASTER_BLUETOOTH)
    set(PROJECT_VER "${STREAMMASTER_VERSION}-bt-dio")
    if(STREAMMASTER_QIO)
        set(PROJECT_VER "${STREAMMASTER_VERSION}-bt-qio")
        if(STREAMMASTER_FLASH80)
            set(PROJECT_VER "${STREAMMASTER_VERSION}-bt-qio80")
        endif()
    endif()
endif()
if(STREAMMASTER_O3)
    set(PROJECT_VER "${STREAMMASTER_VERSION}-o3")
endif()
if(STREAMMASTER_HOT_IRAM)
    if(NOT STREAMMASTER_BLUETOOTH OR NOT STREAMMASTER_QIO OR NOT STREAMMASTER_FLASH80)
        message(FATAL_ERROR "IRAM comparison requires Bluetooth QIO80 baseline")
    endif()
    set(PROJECT_VER "${STREAMMASTER_VERSION}-bt-qio80-iram")
endif()
if(STREAMMASTER_BOARD STREQUAL "generic")
    set(PROJECT_VER "${STREAMMASTER_VERSION}-generic")
    if(STREAMMASTER_QIO AND STREAMMASTER_FLASH80)
        set(PROJECT_VER "${STREAMMASTER_VERSION}-generic-qio80")
    endif()
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(streammaster_onju_v3)
if(CONFIG_BT_BLUEDROID_ENABLED)
    if(NOT STREAMMASTER_BLUETOOTH OR CONFIG_BT_CONTROLLER_ENABLED OR CONFIG_USB_HOST_HUB_MULTI_LEVEL)
        message(FATAL_ERROR "USB Bluetooth requires STREAMMASTER_BLUETOOTH=ON, host-only mode and a single hub")
    endif()
    # Eight host channels: hub EP0 + PSP EP0/IN/OUT + HCI EP0/INT/IN/OUT.
    # Generate one reviewed hub source; never modify the shared SDK checkout.
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/hub_poll_build.py")
    execute_process(COMMAND "${PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/hub_poll_build.py"
        "$ENV{IDF_PATH}/components/usb/ext_hub.c" "${CMAKE_BINARY_DIR}/sm_ext_hub.c"
        COMMAND_ERROR_IS_FATAL ANY)
    idf_component_get_property(sm_usb_lib usb COMPONENT_LIB)
    get_target_property(sm_usb_sources ${sm_usb_lib} SOURCES)
    list(FILTER sm_usb_sources EXCLUDE REGEX "(^|/)ext_hub\\.c$")
    set_property(TARGET ${sm_usb_lib} PROPERTY SOURCES ${sm_usb_sources} "${CMAKE_BINARY_DIR}/sm_ext_hub.c")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/hid_restore_build.py")
    execute_process(COMMAND "${PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/hid_restore_build.py"
        "$ENV{IDF_PATH}/components/bt/host/bluedroid/btc/profile/std/hid/btc_hh.c" "${CMAKE_BINARY_DIR}/sm_btc_hh.c"
        COMMAND_ERROR_IS_FATAL ANY)
    idf_component_get_property(sm_bt_lib bt COMPONENT_LIB)
    get_target_property(sm_bt_sources ${sm_bt_lib} SOURCES)
    list(FILTER sm_bt_sources EXCLUDE REGEX "(^|/)btc_hh\\.c$")
    set_property(TARGET ${sm_bt_lib} PROPERTY SOURCES ${sm_bt_sources} "${CMAKE_BINARY_DIR}/sm_btc_hh.c")
endif()
if(STREAMMASTER_BLUETOOTH AND NOT CONFIG_BT_BLUEDROID_ENABLED)
    message(FATAL_ERROR "Bluetooth build requires sdkconfig.bluetooth")
endif()
if(STREAMMASTER_QIO AND NOT CONFIG_ESPTOOLPY_FLASHMODE_QIO)
    message(FATAL_ERROR "QIO build requires sdkconfig.qio")
endif()
if(STREAMMASTER_BOARD STREQUAL "onju-v3" AND NOT IDF_TARGET STREQUAL "esp32s3")
    message(FATAL_ERROR "Onju V3 requires ESP32-S3; use STREAMMASTER_BOARD=generic for S2")
endif()
if(NOT IDF_TARGET STREQUAL "esp32s3" AND NOT IDF_TARGET STREQUAL "esp32s2")
    message(FATAL_ERROR "StreamMaster supports only ESP32-S3 and ESP32-S2 targets")
endif()
if(STREAMMASTER_O3)
    idf_build_set_property(COMPILE_OPTIONS "-O3" APPEND)
    # IDF 5.5.1 semihost readdir and RTC clocks trigger uninitialized errors at O3.
    # Keep these unrelated sources at their supported level; do not hide
    # warnings or patch the vendored SDK for this comparison.
    idf_component_get_property(streammaster_vfs vfs COMPONENT_LIB)
    set_property(SOURCE "$ENV{IDF_PATH}/components/vfs/vfs_semihost.c"
        TARGET_DIRECTORY ${streammaster_vfs} APPEND PROPERTY COMPILE_OPTIONS -O2)
    idf_component_get_property(streammaster_hw esp_hw_support COMPONENT_LIB)
    set_property(SOURCE "$ENV{IDF_PATH}/components/esp_hw_support/port/esp32s3/rtc_clk.c"
        TARGET_DIRECTORY ${streammaster_hw} APPEND PROPERTY COMPILE_OPTIONS -O2)
endif()
